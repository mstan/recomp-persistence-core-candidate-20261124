cmake_minimum_required(VERSION 3.24)
project(recomp_persistence_core VERSION 0.1.0 LANGUAGES C CXX)
include(CTest)
find_package(Threads REQUIRED)
find_package(ZLIB 1.2.9 REQUIRED)
add_library(recomp_persistence_core STATIC
  src/state_envelope.cpp src/sha256.cpp src/atomic_file.c
  src/native_writer.c src/worker.cpp src/slot_store.cpp)
add_library(recomp-persistence-core::core ALIAS recomp_persistence_core)
target_include_directories(recomp_persistence_core PUBLIC include)
target_compile_features(recomp_persistence_core PUBLIC cxx_std_17)
target_compile_features(recomp_persistence_core PRIVATE c_std_11)
target_link_libraries(recomp_persistence_core PRIVATE Threads::Threads ZLIB::ZLIB)
if(MSVC)
  target_compile_options(recomp_persistence_core PRIVATE /W4)
else()
  target_compile_options(recomp_persistence_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
  foreach(scenario envelope native slots)
    add_executable(persistence_${scenario}_test tests/${scenario}.cpp)
    target_link_libraries(persistence_${scenario}_test PRIVATE recomp_persistence_core)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${scenario}")
    add_test(NAME persistence_${scenario} COMMAND persistence_${scenario}_test)
    set_tests_properties(persistence_${scenario} PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/${scenario}")
  endforeach()
  add_executable(persistence_c_test tests/c_api.c)
  target_link_libraries(persistence_c_test PRIVATE recomp_persistence_core)
  add_test(NAME persistence_c_api COMMAND persistence_c_test)
  if(WIN32)
    add_executable(persistence_legacy_atomic_test tests/legacy_atomic.c src/atomic_file.c)
    target_include_directories(persistence_legacy_atomic_test PRIVATE include)
    target_compile_features(persistence_legacy_atomic_test PRIVATE c_std_11)
    target_compile_definitions(persistence_legacy_atomic_test PRIVATE RECOMP_ATOMIC_TEST_LEGACY_RENAME=1)
    file(MAKE_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/legacy")
    add_test(NAME persistence_legacy_atomic COMMAND persistence_legacy_atomic_test)
    set_tests_properties(persistence_legacy_atomic PROPERTIES WORKING_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}/fixtures/legacy")
  endif()
endif()
